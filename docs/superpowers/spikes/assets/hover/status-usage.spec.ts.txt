import { cp, mkdtemp, writeFile, rm, symlink } from "node:fs/promises";
import { tmpdir } from "node:os";
import path from "node:path";
import { test, expect } from "../support/fixtures";
import { seedWorkspace } from "../support/helpers/seed-client";
import { buildAgentRoute } from "../support/helpers/mock-agent";
import { pluginRequirements } from "../support/helpers/plugin-fixture";
import { connectNewWorkspaceDaemonClient } from "../support/helpers/new-workspace";

test("attributed usage restores account and all five rows across hover, press and view modes", async ({
  page,
}) => {
  test.setTimeout(180_000);
  const source = process.env.CCSTATUSLINE_HOVER_SOURCE;
  if (!source) throw new Error("Set CCSTATUSLINE_HOVER_SOURCE to the plugin integration checkout");
  const workspace = await seedWorkspace({
    repoPrefix: "status-usage-",
    title: "Usage attribution",
  });
  const client = await connectNewWorkspaceDaemonClient({ ownProjects: false });
  const config = await client.getDaemonConfig();
  const directory = await mkdtemp(path.join(tmpdir(), "status-usage-fixture-"));
  try {
    for (const entry of ["index.client.tsx", "client", "shared"])
      await cp(path.join(source, entry), path.join(directory, entry), { recursive: true });
    await symlink(path.join(source, "node_modules"), path.join(directory, "node_modules"), "dir");
    await writeFile(
      path.join(directory, "paseo-plugin.json"),
      JSON.stringify({ id: "paseo-ccstatusline", requirements: pluginRequirements }),
    );
    await writeFile(
      path.join(directory, "index.server.ts"),
      `
      import { z } from "zod";
      import { settingsReadRpc, factsReadRpc } from "./shared/rpc";
      import { parseSettings } from "./shared/settings";
      export default function contribute(server) {
        server.registerUsageSource({ id: "mock", label: "Mock source", input: z.object({ key: z.string() }),
          discover: async (scope) => {
            const keys = scope.kind === "global" ? ["a", "b"] : scope.provider === "mock" ? ["a"] : [];
            return keys.map(key => ({key, label: "Account " + key.toUpperCase(), input: {key}}));
          },
          fetch: async () => ({status: "available", windows: [
            {id: "five_hour", label: "Session", usedPct: 23}, {id: "weekly", label: "Weekly", usedPct: 47},
          ]}),
        });
        server.handle(settingsReadRpc, () => ({ok: true, path: "/fixture/settings", settings: parseSettings({lines: [
          [{type: "model"}], [{type: "current-working-dir"}], [{type: "free-memory"}],
          [{type: "session-usage"}, {type: "weekly-usage"}], [{type: "git-branch"}],
        ]})}));
        server.handle(factsReadRpc, () => ({branch: "fixture-main", memory: {used: 1024, total: 2048}, outputStyle: null}));
        return () => {};
      }
    `,
    );
    const agent = await workspace.client.createAgent({
      provider: "mock",
      cwd: workspace.repoPath,
      workspaceId: workspace.workspaceId,
      title: "Attributed agent",
      model: "ten-second-stream",
      modeId: "load-test",
    });
    await client.patchDaemonConfig({ pluginsEnabled: true });
    await client.installDirectoryPlugin(directory);
    const global = await client.listProviderUsage();
    expect(global.providers.filter((entry) => entry.providerId === "mock")).toHaveLength(2);
    const scoped = await client.listUsageReports({ agentId: agent.id });
    expect(
      scoped.reports
        .filter((entry) => entry.sourceId === "mock")
        .map((entry) => entry.account.label),
    ).toEqual(["Account A"]);
    await page.setViewportSize({ width: 1440, height: 1000 });
    await page.goto(buildAgentRoute(workspace.workspaceId, agent.id));
    const pill = page.getByRole("button").filter({ hasText: / · 23% · 47%/ });
    await expect(pill).toHaveCount(1);
    await pill.hover();
    const rows = page.getByTestId("ccstatusline-rows");
    await expect(page.getByText("Usage: Account A", { exact: true })).toBeVisible();
    await expect(page.getByText("Usage: Account B", { exact: true })).toHaveCount(0);
    async function assertRowCount() {
      await expect
        .poll(() =>
          rows.evaluate((element) => {
            const column = element.firstElementChild?.firstElementChild;
            return column?.children.length;
          }),
        )
        .toBe(5);
      await expect
        .poll(() =>
          rows.evaluate((element) => {
            const column = element.firstElementChild?.firstElementChild;
            const last = column?.lastElementChild;
            return (
              !!last &&
              last.getBoundingClientRect().bottom <= element.getBoundingClientRect().bottom + 1
            );
          }),
        )
        .toBe(true);
    }
    async function assertRows() {
      await assertRowCount();
      await expect(rows.getByText("5h: 23.0% | Weekly: 47.0%", { exact: true })).toBeVisible();
    }
    await assertRows();
    await page.getByRole("button", { name: "Colored view", exact: true }).click();
    await expect(rows.getByText("5h: 23.0%", { exact: true })).toBeVisible();
    await expect(rows.getByText("Weekly: 47.0%", { exact: true })).toBeVisible();
    await assertRowCount();
    await page.mouse.move(0, 0);
    await expect(rows).toHaveCount(0);
    await pill.click();
    await expect(rows).toHaveCount(1);
    await expect(page.getByText("Usage: Account A", { exact: true })).toBeVisible();
    await expect(page.getByRole("button", { name: "Colored view", exact: true })).toHaveAttribute(
      "aria-selected",
      "true",
    );
    await assertRowCount();
    await page.getByRole("button", { name: "Text view", exact: true }).click();
    await assertRows();
    await page.keyboard.press("Escape");
    await page.mouse.move(0, 0);
    await page.setViewportSize({ width: 390, height: 844 });
    await pill.click();
    await expect(page.getByText("Usage: Account A", { exact: true })).toBeVisible();
    await assertRows();
  } finally {
    await client.removePlugin("paseo-ccstatusline");
    await client.patchDaemonConfig({ pluginsEnabled: config.config.pluginsEnabled });
    await client.close();
    await workspace.cleanup();
    await rm(directory, { recursive: true, force: true });
  }
});
